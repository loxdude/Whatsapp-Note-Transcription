cmake_minimum_required(VERSION 3.22.1)
project(voicenote_jni LANGUAGES C CXX)

set(WHISPER_CPP_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../../third_party/whisper.cpp")
set(VULKAN_HEADERS_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../../third_party/Vulkan-Headers-vulkan-sdk-1.4.350.1")
set(SPIRV_HEADERS_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../../third_party/SPIRV-Headers-vulkan-sdk-1.4.350.1")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(WHISPER_CURL OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN ON CACHE BOOL "" FORCE)

if(ANDROID)
    set(Vulkan_INCLUDE_DIR "${VULKAN_HEADERS_DIR}/include" CACHE PATH "" FORCE)
    find_library(ANDROID_VULKAN_LIBRARY vulkan REQUIRED)
    set(Vulkan_LIBRARY "${ANDROID_VULKAN_LIBRARY}" CACHE FILEPATH "" FORCE)
    if(CMAKE_HOST_WIN32)
        set(Vulkan_GLSLC_EXECUTABLE "${ANDROID_NDK}/shader-tools/windows-x86_64/glslc.exe" CACHE FILEPATH "" FORCE)
    endif()
endif()

set(SPIRV_HEADERS_ENABLE_TESTS OFF CACHE BOOL "" FORCE)
set(SPIRV_HEADERS_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory("${SPIRV_HEADERS_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/spirv-headers")
include_directories("${SPIRV_HEADERS_DIR}/include")

add_subdirectory("${WHISPER_CPP_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/whisper.cpp")

add_library(voicenote_jni SHARED native_parakeet_bridge.cpp)
target_compile_options(voicenote_jni PRIVATE -O3 -march=armv8.2-a+fp16)
target_include_directories(voicenote_jni PRIVATE "${WHISPER_CPP_DIR}/include")
target_link_libraries(voicenote_jni PRIVATE parakeet android log)
