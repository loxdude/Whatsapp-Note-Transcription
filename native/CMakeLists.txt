cmake_minimum_required(VERSION 3.16)
project(voicenotes_opus C)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_C_STANDARD 99)
set(OPUS_BUILD_SHARED_LIBRARY OFF CACHE BOOL "" FORCE)
set(OPUS_BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(OPUS_BUILD_PROGRAMS OFF CACHE BOOL "" FORCE)
set(OPUS_INSTALL_PKG_CONFIG_MODULE OFF CACHE BOOL "" FORCE)
set(OPUS_INSTALL_CMAKE_CONFIG_MODULE OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
add_subdirectory("${DEPS}/opus-1.5.2" opus)
add_subdirectory("${DEPS}/libogg-1.3.5" ogg)
add_library(opusfile STATIC
    "${DEPS}/opusfile-0.12/src/info.c" "${DEPS}/opusfile-0.12/src/internal.c"
    "${DEPS}/opusfile-0.12/src/opusfile.c" "${DEPS}/opusfile-0.12/src/stream.c")
target_include_directories(opusfile PUBLIC "${DEPS}/opusfile-0.12/include"
    "${DEPS}/opus-1.5.2/include" "${DEPS}/libogg-1.3.5/include" "${CMAKE_CURRENT_BINARY_DIR}/ogg/include")
target_link_libraries(opusfile PUBLIC opus ogg m)
add_library(voicenotes_opus SHARED decoder.c)
target_link_libraries(voicenotes_opus PRIVATE opusfile)
target_compile_options(voicenotes_opus PRIVATE -Wall -Wextra -Werror)
if(ANDROID)
    target_sources(voicenotes_opus PRIVATE jni.c)
    target_link_options(voicenotes_opus PRIVATE "-Wl,-z,max-page-size=16384")
endif()
